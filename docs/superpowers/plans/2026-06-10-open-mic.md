# Open-Mic Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `/open-mic` — a full-viewport 3D live show page where Deax (a realistic Ready Player Me avatar) hosts a personality-driven voice conversation on a virtual nightclub stage, rendering portfolio sections on a giant LED screen when triggered.

**Architecture:** Fully isolated page under `src/app/open-mic/` and `src/components/open-mic/`. React Three Fiber canvas covers the full viewport — no traditional HTML layout. A `useReducer`-based phase machine (`landing → connecting → live → ended`) drives the scene. LiveKit (raw `livekit-client` SDK) handles voice; data channel messages on topic `open-mic-events` drive section display and avatar animation.

**Tech Stack:** Next.js App Router · TypeScript · `@react-three/fiber` · `@react-three/drei` · `three` · `livekit-client` (already installed) · `portfolio.json` data · Jest (existing setup)

```typescript
export type OpenMicPhase = 'landing' | 'connecting' | 'live' | 'ended'
export type ActiveSection = 'skills' | 'projects' | 'about' | 'experience' | 'contact' | null
export type DeaxAnimationState = 'idle' | 'listening' | 'thinking' | 'speaking' | 'presenting'
export type DeaxEmotion = 'neutral' | 'excited' | 'proud' | 'playful' | 'thoughtful' | 'attentive'
export type AgentState = 'listening' | 'thinking' | 'speaking'

export interface OpenMicState {
  phase: OpenMicPhase
  activeSection: ActiveSection
  agentState: AgentState
  emotion: DeaxEmotion
}

export type OpenMicAction =
  | { type: 'START' }
  | { type: 'CONNECTED' }
  | { type: 'END' }
  | { type: 'RESET' }
  | { type: 'SET_SECTION'; section: ActiveSection }
  | { type: 'CLEAR_SECTION' }
  | { type: 'SET_AGENT_STATE'; state: AgentState }
  | { type: 'SET_EMOTION'; emotion: DeaxEmotion }

export type DataChannelMessage =
  | { type: 'agent_state'; state: AgentState }
  | { type: 'action'; action: 'show_section'; section: ActiveSection }
  | { type: 'action'; action: 'hide_section' }
  | { type: 'action'; action: 'end_show' }
  | { type: 'emotion'; state: DeaxEmotion }

export const initialOpenMicState: OpenMicState = {
  phase: 'landing',
  activeSection: null,
  agentState: 'listening',
  emotion: 'neutral',
}

export function openMicReducer(state: OpenMicState, action: OpenMicAction): OpenMicState {
  switch (action.type) {
    case 'START':           return { ...state, phase: 'connecting' }
    case 'CONNECTED':       return { ...state, phase: 'live' }
    case 'END':             return { ...state, phase: 'ended', activeSection: null }
    case 'RESET':           return initialOpenMicState
    case 'SET_SECTION':     return { ...state, activeSection: action.section }
    case 'CLEAR_SECTION':   return { ...state, activeSection: null }
    case 'SET_AGENT_STATE': return { ...state, agentState: action.state }
    case 'SET_EMOTION':     return { ...state, emotion: action.emotion }
    default: return state
  }
}
```

```typescript
import type { DataChannelMessage, AgentState, DeaxEmotion, ActiveSection } from '../types'

export const OPEN_MIC_TOPIC = 'open-mic-events'

const VALID_AGENT_STATES = new Set<AgentState>(['listening', 'thinking', 'speaking'])
const VALID_EMOTIONS = new Set<DeaxEmotion>(['neutral', 'excited', 'proud', 'playful', 'thoughtful', 'attentive'])
const VALID_SECTIONS = new Set<string>(['skills', 'projects', 'about', 'experience', 'contact'])

export function parseDataChannelMessage(payload: Uint8Array): DataChannelMessage | null {
  try {
    const text = new TextDecoder().decode(payload)
    const msg = JSON.parse(text)

    if (msg.type === 'agent_state' && VALID_AGENT_STATES.has(msg.state)) {
      return { type: 'agent_state', state: msg.state }
    }
    if (msg.type === 'action') {
      if (msg.action === 'show_section' && VALID_SECTIONS.has(msg.section)) {
        return { type: 'action', action: 'show_section', section: msg.section as ActiveSection }
      }
      if (msg.action === 'hide_section') return { type: 'action', action: 'hide_section' }
      if (msg.action === 'end_show') return { type: 'action', action: 'end_show' }
    }
    if (msg.type === 'emotion' && VALID_EMOTIONS.has(msg.state)) {
      return { type: 'emotion', state: msg.state }
    }
    return null
  } catch {
    return null
  }
}
```

```tsx
'use client'
import { useRef } from 'react'
import { useFrame, useThree } from '@react-three/fiber'
import * as THREE from 'three'
import type { OpenMicPhase } from '../types'

const LANDING_POS = new THREE.Vector3(0, 1.5, 13)
const LIVE_POS    = new THREE.Vector3(0, 1.5, 7)
const LOOK_AT     = new THREE.Vector3(0.5, 0.5, 0)

export function CameraRig({ phase }: { phase: OpenMicPhase }) {
  const { camera } = useThree()
  const target = phase === 'landing' || phase === 'connecting' ? LANDING_POS : LIVE_POS

  useFrame(() => {
    // lerp factor 0.025 → ~3s smooth dolly when phase changes to 'live'
    camera.position.lerp(target, 0.025)
    camera.lookAt(LOOK_AT)
  })

  return null
}
```

- [ ] **Step 1: Create the hook**

`src/components/open-mic/voice/useOpenMicVoice.ts`:
```typescript
import { useRef, useCallback, useEffect } from 'react'
import { Room, RoomEvent } from 'livekit-client'
import { parseDataChannelMessage, OPEN_MIC_TOPIC } from './OpenMicDataChannel'
import type { DataChannelMessage } from '../types'

export function useOpenMicVoice(
  onMessage: (msg: DataChannelMessage) => void,
  onDisconnected: () => void
) {
  const roomRef = useRef<Room | null>(null)
  // Keep handler refs current so connect() closure never goes stale
  const onMessageRef = useRef(onMessage)
  const onDisconnectedRef = useRef(onDisconnected)
  onMessageRef.current = onMessage
  onDisconnectedRef.current = onDisconnected

  const connect = useCallback(async () => {
    const res = await fetch(
      `${process.env.NEXT_PUBLIC_VOICE_AGENT_URL}/api/open-mic/token/`,
      { method: 'POST' }
    )
    if (!res.ok) throw new Error(`Token fetch failed: ${res.status}`)
    // Backend returns { token, url } — verify field names against backend endpoint
    const { token, url } = await res.json()

    const room = new Room()
    roomRef.current = room

    room.on(RoomEvent.DataReceived, (payload, _participant, _kind, topic) => {
      if (topic !== OPEN_MIC_TOPIC) return
      const msg = parseDataChannelMessage(payload)
      if (msg) onMessageRef.current(msg)
    })

    room.on(RoomEvent.Disconnected, () => onDisconnectedRef.current())

    await room.connect(url, token)
    await room.localParticipant.setMicrophoneEnabled(true)
    // Enables autoplay of remote audio tracks (agent voice)
    // livekit-client v2 handles remote audio automatically after connect + user gesture
  }, [])

  const disconnect = useCallback(() => {
    roomRef.current?.disconnect()
    roomRef.current = null
  }, [])

  // Cleanup on unmount
  useEffect(() => () => { roomRef.current?.disconnect() }, [])

  return { connect, disconnect }
}
```

> **Note on audio:** If the agent's voice is silent, call `await roomRef.current?.startAudio()` after `room.connect()`. This may be needed if the user hasn't interacted with the page before the autoplay policy fires.

```tsx
'use client'
import { useReducer, useCallback } from 'react'
import { openMicReducer, initialOpenMicState } from './types'
import { StageScene } from './scene/StageScene'
import { LandingOverlay } from './ui/LandingOverlay'
import { useOpenMicVoice } from './voice/useOpenMicVoice'
import type { DataChannelMessage } from './types'

export default function OpenMicStage() {
  const [state, dispatch] = useReducer(openMicReducer, initialOpenMicState)

  const handleMessage = useCallback((msg: DataChannelMessage) => {
    if (msg.type === 'agent_state')
      dispatch({ type: 'SET_AGENT_STATE', state: msg.state })
    else if (msg.type === 'emotion')
      dispatch({ type: 'SET_EMOTION', emotion: msg.state })
    else if (msg.type === 'action' && msg.action === 'show_section')
      dispatch({ type: 'SET_SECTION', section: msg.section })
    else if (msg.type === 'action' && msg.action === 'hide_section')
      dispatch({ type: 'CLEAR_SECTION' })
    else if (msg.type === 'action' && msg.action === 'end_show')
      dispatch({ type: 'END' })
  }, [])

  const handleDisconnected = useCallback(() => {
    dispatch({ type: 'RESET' })
  }, [])

  const { connect, disconnect } = useOpenMicVoice(handleMessage, handleDisconnected)

  const handleStart = useCallback(async () => {
    dispatch({ type: 'START' })
    try {
      await connect()
      dispatch({ type: 'CONNECTED' })
    } catch (err) {
      console.error('[open-mic] connection failed', err)
      dispatch({ type: 'RESET' })
    }
  }, [connect])

  const showLanding = state.phase === 'landing' || state.phase === 'connecting'

  return (
    <div className="om-root">
      <StageScene
        phase={state.phase}
        activeSection={state.activeSection}
        animState="idle"
        emotion={state.emotion}
        sectionContent={null}
      />
      {showLanding && (
        <LandingOverlay
          onStart={handleStart}
          connecting={state.phase === 'connecting'}
        />
      )}
    </div>
  )
}
```

```tsx
'use client'
import { useRef, useEffect } from 'react'
import { useGLTF, useAnimations } from '@react-three/drei'
import type { Group } from 'three'
import type { DeaxAnimationState, DeaxEmotion } from '../types'

// Maps animation state + emotion to an animation clip name from the GLB.
// Clip names must match what's embedded in deax.glb (verify in Blender or gltf.report).
function resolveClipName(animState: DeaxAnimationState, emotion: DeaxEmotion): string {
  if (animState === 'presenting') return 'Pointing'
  if (animState === 'thinking') return 'Thinking'
  if (animState === 'speaking') return emotion === 'excited' ? 'Talking_2' : 'Talking_1'
  if (animState === 'listening') return 'Listening'
  return 'Idle'
}

interface DeaxCharacterProps {
  animState: DeaxAnimationState
  emotion: DeaxEmotion
}

export function DeaxCharacter({ animState, emotion }: DeaxCharacterProps) {
  const groupRef = useRef<Group>(null)
  const { scene, animations } = useGLTF('/models/deax.glb')
  const { actions } = useAnimations(animations, groupRef)

  useEffect(() => {
    if (!actions || Object.keys(actions).length === 0) return
    const clipName = resolveClipName(animState, emotion)
    // Fall back to first available clip if named clip not found
    const action = actions[clipName] ?? actions[Object.keys(actions)[0]]
    action.reset().fadeIn(0.3).play()
    return () => { action.fadeOut(0.3) }
  }, [animState, emotion, actions])

  return (
    // Position: center-right, waist level at y=-2 (camera looks at y=0.5)
    <group ref={groupRef} position={[1.5, -2, 0]} scale={1.8}>
      <primitive object={scene} />
    </group>
  )
}

useGLTF.preload('/models/deax.glb')
```

```tsx
'use client'
import { useReducer, useCallback, useEffect } from 'react'
import { openMicReducer, initialOpenMicState } from './types'
import { StageScene } from './scene/StageScene'
import { LandingOverlay } from './ui/LandingOverlay'
import { MobileBlocker } from './ui/MobileBlocker'
import { useOpenMicVoice } from './voice/useOpenMicVoice'
import { SkillsSection } from './sections/SkillsSection'
import { ProjectsSection } from './sections/ProjectsSection'
import { AboutSection } from './sections/AboutSection'
import { ExperienceSection } from './sections/ExperienceSection'
import { ContactSection } from './sections/ContactSection'
import type { DataChannelMessage, ActiveSection, DeaxAnimationState, OpenMicState } from './types'

function deriveAnimState(state: OpenMicState): DeaxAnimationState {
  if (state.activeSection) return 'presenting'
  if (state.agentState === 'speaking') return 'speaking'
  if (state.agentState === 'thinking') return 'thinking'
  return 'listening'
}

function getSectionContent(section: ActiveSection): React.ReactNode {
  switch (section) {
    case 'skills':     return <SkillsSection />
    case 'projects':   return <ProjectsSection />
    case 'about':      return <AboutSection />
    case 'experience': return <ExperienceSection />
    case 'contact':    return <ContactSection />
    default:           return null
  }
}

export default function OpenMicStage() {
  const [state, dispatch] = useReducer(openMicReducer, initialOpenMicState)

  const handleMessage = useCallback((msg: DataChannelMessage) => {
    if (msg.type === 'agent_state')
      dispatch({ type: 'SET_AGENT_STATE', state: msg.state })
    else if (msg.type === 'emotion')
      dispatch({ type: 'SET_EMOTION', emotion: msg.state })
    else if (msg.type === 'action' && msg.action === 'show_section')
      dispatch({ type: 'SET_SECTION', section: msg.section })
    else if (msg.type === 'action' && msg.action === 'hide_section')
      dispatch({ type: 'CLEAR_SECTION' })
    else if (msg.type === 'action' && msg.action === 'end_show')
      dispatch({ type: 'END' })
  }, [])

  const handleDisconnected = useCallback(() => {
    dispatch({ type: 'RESET' })
  }, [])

  const { connect, disconnect } = useOpenMicVoice(handleMessage, handleDisconnected)

  const handleStart = useCallback(async () => {
    dispatch({ type: 'START' })
    try {
      await connect()
      dispatch({ type: 'CONNECTED' })
    } catch (err) {
      console.error('[open-mic] connection failed', err)
      dispatch({ type: 'RESET' })
    }
  }, [connect])

  // Auto-reset to landing after 'ended' phase (2s pause so Deax can finish his sign-off)
  useEffect(() => {
    if (state.phase !== 'ended') return
    const t = setTimeout(() => {
      disconnect()
      dispatch({ type: 'RESET' })
    }, 2000)
    return () => clearTimeout(t)
  }, [state.phase, disconnect])

  const showLanding = state.phase === 'landing' || state.phase === 'connecting'

  return (
    <div className="om-root">
      <MobileBlocker />
      <StageScene
        phase={state.phase}
        activeSection={state.activeSection}
        animState={deriveAnimState(state)}
        emotion={state.emotion}
        sectionContent={getSectionContent(state.activeSection)}
      />
      {showLanding && (
        <LandingOverlay
          onStart={handleStart}
          connecting={state.phase === 'connecting'}
        />
      )}
    </div>
  )
}
```
